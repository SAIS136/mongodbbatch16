exe 1 (multiply,subtract)

db.programmers.aggregate([
    {
        $project:{
            name:1,
            salary:1,
            taxAmount:{
                $multiply:["$salary","$tax"]
            },
            GrandSalary:{
                $subtract:["$salary",{
                    $multiply:["$salary","$tax"]
                }]
            }
        }
    }
])



exe 2 (multiply,divide)

db.programmers.aggregate([
    {
        $project:{
            name:1,
            salary:1,
            workedHours:1,
            weeklyHours:{$multiply:["$workedHours",4]}, // Approximate Monthly
            hourlyRate:{
                $divide:[
                    "$salary",
                    {$multiply:["$workedHours",4]}
                ]   // Annual salary / Annual hours
            }
        }
    }
])



exe 3 (round,ceil,floors)

db.items.aggregate([

            name:1,
            originalPrice:"$price",
            roundedPrice:{$round:"$price"},
            ceilPrice:{$ceil:"$price"},
            floorPrice:{$floor:"$price"}
        }
    }
])



exe 4 (pow,sqrt)

db.items.aggregate([
    {
        $project:{
            name:1,
            weight:1,
            weightPow:{

            weightSquare:{
                $sqrt:"$weight"
            }
        }
    }
])

db.items.aggregate([

db.items.aggregate([

    }
