    {
        $project:{
            name:1,
            year:{$year:"$start"},
            month:{$month:"$start"},
            year:{$dayOfMonth:"$start"}
        }
    }
])

    {
        $project:{
            name:1,
            time:{
                hour:{$hour:"$start"},
                minute:{$minute:"$start"},
                second:{$second:"$start"}
            }
        }
    }
])



exe 3

db.events.aggregate([
    {
        $project:{
            name:1,
            startLocal:{
                $dateToString:{
                    format:"%Y-%m-%d %H:%M:%S",
                    date:"$start",
                    timezone:"Asia/Yangon"
                }
            }
        }
    }
])



exe 4

db.events.aggregate([
    {
        $project:{
            name:1,
            deadlineDate:{
                $dateFromString:{
                    format:"%Y-%m-%d %H:%M:%S",
                    dateString:"$dateString",
                    timezone:"$tz"
                }
            }
        }
    }
])




exe 5 (add 10 minute)

db.events.aggregate([
    {
        $project:{
            name:1,
            start:1,
            plus10min:{
                $dateAdd:{
                    startDate:"$start",
                    unit:"minute", // minute,month..
                    amount:10
                }
            }
        }
    }
])




exe 6 (subtract 1 month)

db.events.aggregate([
    {
        $project:{
            name:1,
            end:1,
            minus1month:{
                $dateSubtract:{
                    startDate:"$end",
                    unit:"month", // minute,month..
                    amount:1
                }
            }
        }
    }
])




exe 7 (subtract 1 month)

db.events.aggregate([
    {
        $project:{
            name:1,
            minDiff:{
                $dateDiff:{
                    startDate:"$start",
                    endDate:"$end",
                    unit:"minute"
                }
            },
            daysDiff:{
                $dateDiff:{
                    startDate:"$start",
                    endDate:"$end",
                    unit:"day"
                }
            },
        }
    }
])
