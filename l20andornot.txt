=> $and { $and : [{condition1},{condition2},...] }

    Implicit $and

    db.consumers.find({
        department:"Engineering",
        isAction:true
    })


    Explicit $and

    db.consumers.find({
        $and:[
            {department:"Engineering"},
            {isAction:true}
        ]
    })


    db.consumers.find({
        $and:[
            {department:"Engineering"},
            {salary:{$gte:700000}},
            {salary:{$lte:950000}}
        ]
    })


    db.consumers.find({
        $and:[
            {department:"Engineering"},
            {isAction:true},
            {location:"Yangon"},
