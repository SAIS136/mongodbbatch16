=> $mergeObjects (combining documents)

exe 1

db.retailsales.aggregate([
    {
        $project:{
            mergeDoc:{
                $mergeObjects:[
                    {name:"$name",category:"$category"},
                    "$details",
                    "$inventory"
                ]
            }
        }
    }
])


exe 2

db.retailsales.aggregate([
    {
        $project:{
            fullProduct:{
                $mergeObjects:[
                    {
                        productId:"$_id",
                        productName:"$name",
                        productCategory:"$category"
                    },
                    "$details",
                    {
                        priceStats:{
                            avgPrice:{$avg:"$price"},
                            minPrice:{$min:"$price"},
                            maxPrice:{$max:"$price"},
                        }
                    },
                    {
                        $cond:{
                            if:{$gt:[{$ifNull:["$inventory.stock",0]},100]},
                            then:{availability:"In Stock"},
                            else:{availability:"Limited"}
                        }
                    }
                ]
            }
        }
    }
])
