# exe1(i) Join posts with users

db.posts.aggregate([
    {
        $lookup:{
            from:"users",
            localField:"userId",
            foreignField:"_id",
            as:"author"
        }
    },
    {
        $project:{
            title:1,
            content:1,
            "author.name":1,
            "author.email":1
        }
    }
])

# exe1(ii) Join posts with users

db.posts.aggregate([
    {
        $lookup:{
            from:"users",
            localField:"userId",
            foreignField:"_id",
            as:"author" // output array field
        }
    },
    {
        $unwind:"$author" // convert array to object
    },
    {
        $project:{
            title:1,
            content:1,
            "author.name":1,
            "author.email":1
        }
    }
])

# exe3(i)

db.posts.aggregate([
    {
        $lookup:{
            from:"users",
            let:{authId:"$userId},
            pipeline:[
                {
                    $match:{
                        $expr:{$eq:["$_id","$$authId"]}
                    }
                }
            ],
            as:"author"
        }
    }
])

db.courses.aggregate([
    {
        $lookup:{
            from:"students",
            localField:"studentIds",
            foreignField:"_id",
            as:"studentEnrolled"
        }
    },
    {
        $project:{
            name:1,
            instructor:1,
