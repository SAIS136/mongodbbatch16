db.runCommand({
    collMod:"products",
    validator:{
        $and:[
            {
                $jsonSchema:{
                bsonType:"object",
                required:["name","category","price","stock","createdAt"],
                additionalProperties:true,
                    properties:{
                    name:{bsonType:"string",minLength:2},
                    category:{bsonType:"string",enum:["fashion","electronics","food"]},
                    price:{bsonType:["int","long","double","decimal"],minimum:0},
                    discountPrice:{bsonType:["int","long","double","decimal"],minimum:0},
                    stock:{bsonType:"int",minimum:0},
                    createdAt:{bsonType:"date"},
                    tags:{
                        bsonType:"array",
                        items:{bsonType:"string"}
                    },
                    isActive:{bsonType:"bool"}
                    }
                }
            },
            {
                $expr:{
                    $or:[
                        {$eq:["$discountPrice",null]},
                        {$lte:["$discountPrice","$price"]}
                    ]
                }
            }
        ]
    },
    validationLevel:"strict",
    validationAction:"error"
})

                    userId:{bsonType:["int","long"]},

                    createdAt:{bsonType:"date"},
                    items:{
                        bsonType:"array",
                        minItems:1,
                        items:{
                            bsonType:"object",
                            required:["productId","name","unitPrice","qty"],
                            properties:{
                                productId:{bsonType:["int","long"]},
                                name:{bsonType:"string"},
                                unitPrice:{bsonType:["int","long","double","decimal"],minimum:0},
                                qty:{bsonType:"int",minimum:1}

            }
    },
    validationLevel:"strict",
    validationAction:"error"
})
