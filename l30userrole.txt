=> Change Role

mongodbcloud123

use proone

    user:"reportuser",
    pwd:"123456789",
    roles:[
        {role:"read",db:"proone"}
    ]
})

    {role:"readWrite",db:"proone"}
])


exit

=> Change Role (ii)

mongosh -u rootadmin -p --authenticationDatabase admin
mongodbcloud123

use proone

db.grantRolesToUser("reportuser",[
    {role:"readWrite",db:"proone"}
])

db.getUsers();

db.revokeRolesFromUser("reportuser",[
    {role:"readWrite",db:"proone"}
])


exit
